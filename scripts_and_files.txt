useful scripts and config files (need sudo)
===========================================
/usr/local/bin/
    av.sh                      # anti virus
    clone/
        clone.sh               # clone the entire hard drive to another
        base_functions.sh      # don't run this (used only by clone.sh)
        fstypes                # text file that maps filesystem names to the command that creates them (don't modify it)
        exclude                # text file that contains dirs/files to exclude from cloning if src does not fit on dst (modifiable)
    cp_kodi_cfg.sh             # copy kodi config files from one user account to another
    useradd/
        useradd.sh             # add a new user; delete destination if it exists
        new_user/              # directory that contains new user data
    userdel.sh                 # delete a user
    upd_dbs_and_keyring.sh     # update pacman DBs, core & extra, and update keyring package in case there's a newer version
    upd_grub_kernel_version.sh # update kernel version in grub.cfg, called from /usr/share /libalpm/hooks/upd_grub_kernel_version.hook
    cpshim.sh                  # copy shim binaries to boot partition, called from /usr/share/libalpm/hooks/cpshim.hook
    wait.sh                    # used in desktop launchers that launch scripts to prevent the script window from closing

/usr/lib/systemd/scripts/
    sync_timedate.sh           # called after wakeup from suspend/hibernate in /etc/systemd/system/sync_timedate.service
    disable_wakeup.sh          # called from /etc/systemd/system/disable_wakeup.service
    pkg_maint.sh               # called from /etc/systemd/system/pkg_maint.service which is called from /etc/systemd/system/pkg_maint.timer

/etc/skel/.bashrc              # useful stuff for all users
/etc/skel/.config/locale.conf  # default language for new user
/etc/sudoers.d/override        # users added to the sudo group don't have to use their password

/etc/systemd/system/lightdm.service.d/override.conf                  # override to display greeter in Greek
/etc/systemd/system/pamac-cleancache.service.d/override.conf         # override to remove pamac build files
/etc/systemd/system/archlinux-keyring-wkd-sync.timer.d/override.conf # override to run timer daily
/etc/systemd/system/pacman-filesdb-refresh.timer.d/override.conf     # override to refresh yay files DB daily
/etc/systemd/sleep.conf.d/override.conf                              # override default sleep, suspend, hibernate settings

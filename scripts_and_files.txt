maintenance scripts
===================
/etc/NetworkManager/dispatcher.d/update-timezone.sh # update the timezone when network interface goes up
/etc/initcpio/post/kernel-sbsign                    # sign linux kernel and update grub.cfg (see /usr/bin/grub_cfg_upd.sh)

/usr/bin/
    cp_kodi_cfg.sh         # copy kodi config files from one user account to another
    grub_cfg_upd.sh        # update grub.cfg when installing/upgrading linux kernels
    pacman_keyring.sh      # update pacman DB and keyring packages if there's an updated keyring package
    wait.sh                # used in desktop launchers that launch scripts to prevent the script window from closing

/usr/lib/systemd/scripts/
    lightdm.sh             # called from /usr/lib/systemd/system/lightdm.service (upd cfg file for intel gfx driver so no Xorg freeze and then run lightdm)
    pkg_maint.sh           # called from /etc/systemd/system/pkg_maint.service which is called from /etc/systemd/system/pkg_maint.timer
                           # (remove orphan packages, clear the package cache and generate new mirrorlist file)

/usr/sbin/
    av.sh                  # anti virus
    clone/
        base_functions.sh  # don't run this (used only by clone.sh) 
        clone.sh           # clone one drive to another
        manpages/
            clone          # man page for clone script
            clone-exclude  # man page for exclude configuration file for clone script 
            manpages.sh    # create man pages for clone script 
    cpshim.sh              # copy shim binaries to boot partition when installing/upgrading shim signed               (see configuration files)
    grub_sbsign.sh         # install and sign grub bootloader and update grub cfg file when installing/upgrading grub (ditto)
    useradd/
        new_user/          # directory that contains new user data
        useradd.sh         # add a new user and delete destination if it exists already
    userdel.sh             # delete a user

configuration files
===================
/etc/clone/README          # point the user to the man pages for clone script
/etc/clone/exclude         # contains dirs/files to exclude from cloning            (used by /usr/sbin/clone/clone.sh above)
/etc/clone/fstypes         # maps filesystem names to the command that creates them (ditto)
/etc/grub.d/09_unrestricted_menu_entries                             # custom grub file to make menu entries unrestricted
/etc/grub.d/39_add_password                                          # custom grub file to add grub user and password for command line and menu entries
/etc/skel/.bashrc                                                    # useful stuff for all users
/etc/skel/.config/locale.conf                                        # default language for new user
/etc/sudoers.d/override                                              # users added to the sudo group don't have to use their password
/etc/systemd/logind.conf.d/override.conf                             # override default user login settings
/etc/systemd/sleep.conf.d/override.conf                              # override default sleep, suspend, hibernate settings
/etc/systemd/system/archlinux-keyring-wkd-sync.timer.d/override.conf # override to run timer daily
/etc/systemd/system/pacman-filesdb-refresh.timer.d/override.conf     # override to refresh pacman DB packages daily
/etc/systemd/system/pamac-cleancache.service.d/override.conf         # override to remove pamac build files
/usr/lib/systemd/system/lightdm.service                              # updated to launch script and set language to Greek
/usr/share/libalpm/hooks/cpshim.hook                                 # hook for script with same name under /usr/sbin
/usr/share/libalpm/hooks/grub_sbsign.hook                            # ditto

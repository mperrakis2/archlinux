   # grub-install
   b. sudo grub-install --modules="$grub_modules" --sbat=/usr/share/grub/sbat.csv --efi-directory="$(bootctl -p)" --recheck --target=x86_64-efi

# check https://wiki.archlinux.org/title/Unified_Extensible_Firmware_Interface/Secure_Boot#shim

2. install shim-signed & sbsigntools packages

3. a. sudo cp /usr/share/shim-signed/mmx64.efi "$(bootctl -p)"/EFI/BOOT/
   b. sudo cp /usr/share/shim-signed/shimx64.efi "$(bootctl -p)"/EFI/BOOT/BOOTX64.EFI
   c. sudo cp /usr/share/shim-signed/shimx64.efi /usr/share/shim-signed/mmx64.efi "$(bootctl -p)"/EFI/"$(uname -n)"/
   d. sudo efibootmgr --unicode --disk /dev/nvme0n1 --part 2 --create --label "shim-$(uname -n)" --loader /EFI/"$(uname -n)"/shimx64.efi

4. a. cd "$(bootctl -p)"/EFI
   Execute once only -> sudo openssl req -newkey rsa:2048 -nodes -keyout MOK.key -new -x509 -sha256 -days 3650 -subj "/CN=my Machine Owner Key/" -out MOK.crt
                     -> sudo openssl x509 -outform DER -in MOK.crt -out MOK.cer
   b. sudo sbsign --key MOK.key --cert MOK.crt --output "$(bootctl -p)"/vmlinuz-linux "$(bootctl -p)"/vmlinuz-linux
   c. sudo sbsign --key MOK.key --cert MOK.crt --output "$(bootctl -p)"/vmlinuz-linux-lts "$(bootctl -p)"/vmlinuz-linux-lts
   d. sudo sbsign --key MOK.key --cert MOK.crt --output "$(bootctl -p)"/EFI/"$(uname -n)"/grubx64.efi "$(bootctl -p)"/EFI/"$(uname -n)"/grubx64.efi
   e. sudo cp "$(bootctl -p)"/EFI/"$(uname -n)"/grubx64.efi "$(bootctl -p)"/EFI/BOOT/
   Execute once only -> Create the following file and make it executable: /etc/initcpio/post/kernel-sbsign (see link above for file contents)

5. a. Reboot and enable Secure Boot. If shim does not find the certificate grubx64.efi is signed with in MokList it will launch MokManager (mmx64.efi).
   b. In MokManager select Enroll key from disk, find MOK.cer and add it to MokList. When done select Continue boot and your boot loader will launch and it will be capable launching any binary signed with your Machine Owner Key.

unified kernel images
=====================
https://wiki.archlinux.org/title/Unified_kernel_image#sbctl

uname -m   # get architecture
bootctl -p # ESP mount dir
bootctl -x # bootloader mount dir
bootctl status --no-pager (dir of grubx64.efi under section 'Boot', dir of BOOTX64.efi under section 'Available')

update /etc/kernel/cmdline # e.g. root=UUID=55e14ea4-8667-4d95-9b49-321c1b166e2b rw loglevel=6 resume=UUID=55e14ea4-8667-4d95-9b49-321c1b166e2b resume_offset=7542784

save cmdline file above and grub.cfg to google drive

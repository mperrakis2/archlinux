unified kernel images
=====================
https://wiki.archlinux.org/title/Unified_kernel_image#sbctl

uname -m   # get architecture
bootctl -x # ESP mount dir
bootctl -x # bootloader mount dir
bootctl status --no-pager (dir of grubx64.efi under section 'Boot', dir of 
                           BOOTX64.efi under section 'Available')

update /etc/kernel/cmdline # e.g. root=UUID=55e14ea4-8667-4d95-9b49-321c1b166e2b
                           #      rw loglevel=6 
                           # resume=UUID=55e14ea4-8667-4d95-9b49-321c1b166e2b
                           # resume_offset=7542784

add menu entries to grub.cfg for the two ukis (below)
update cmdline file above and grub.cfg on github

sudo sbctl bundle --save "$(bootctl -x)"/EFI/arch/uki-linux-x86_64.efi
sudo sbctl bundle --save "$(bootctl -x)"/EFI/arch/uki-linux-lts-x86_64.efi \
                  --kernel-img "$(bootctl -x)"/vmlinuz-linux-lts \
                  --initramfs "$(bootctl -x)"/initramfs-linux-lts.img
sudo sbctl generate-bundles [--sign]
